cmake_minimum_required(VERSION 3.16)
project(MarketSignalLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(market_sim simulator/main.cpp)
target_compile_options(market_sim PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME accounting_contract COMMAND ${Python3_EXECUTABLE} -m pytest -q tests/test_simulator.py)
set_tests_properties(accounting_contract PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} ENVIRONMENT "MARKET_SIM=$<TARGET_FILE:market_sim>")
